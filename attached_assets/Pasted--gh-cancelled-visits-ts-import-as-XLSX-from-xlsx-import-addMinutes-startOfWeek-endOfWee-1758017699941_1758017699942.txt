// gh-cancelled-visits.ts
import * as XLSX from 'xlsx';
import { addMinutes, startOfWeek, endOfWeek, format as fmt } from 'date-fns';

const HEADERS = {
  cancel: 'Cancellation Description',
  start: 'Service Requirement Start Date And Time',   // datetime or Excel serial
  duration: 'Service Requirement Duration',          // HOURS (decimal)
  staff: ['Planned Employee Name','Template Employee Name','Actual Employee Name'],
};

function toDate(v: any): Date | undefined {
  if (v instanceof Date && !isNaN(+v)) return v;
  if (typeof v === 'number') {
    const d = XLSX.SSF.parse_date_code(v);
    if (d) return new Date(d.y, (d.m ?? 1)-1, d.d ?? 1, d.H ?? 0, d.M ?? 0, Math.floor(d.S ?? 0));
  }
  const t = new Date(String(v)); return isNaN(+t) ? undefined : t;
}

function norm(s: string) { return s.trim().toLowerCase().replace(/\s+/g,' '); }

export function cancelledWindowsFromGH(
  workbookBuf: Buffer,
  selectedDate: Date,
  minMinutes = 60
): Map<string, string> {
  const wb = XLSX.read(workbookBuf, { type: 'buffer' });
  const sheet = wb.SheetNames.find(n => n.toLowerCase().includes('data')) ?? wb.SheetNames[0];
  const rows = XLSX.utils.sheet_to_json<Record<string, any>>(wb.Sheets[sheet], { raw: true });

  const wkStart = startOfWeek(selectedDate, { weekStartsOn: 1 });
  const wkEnd   = endOfWeek(selectedDate, { weekStartsOn: 1 });

  const tmp = new Map<string, string[]>();

  for (const r of rows) {
    const status = String(r[HEADERS.cancel] ?? '').toLowerCase();
    if (!status.includes('cancel')) continue;

    const start = toDate(r[HEADERS.start]);
    const durH = Number(r[HEADERS.duration]);
    if (!start || !isFinite(durH)) continue;

    const minutes = Math.round(durH * 60);
    if (minutes < minMinutes || start < wkStart || start > wkEnd) continue;

    const staff = HEADERS.staff.map(k => r[k]).find(v => v && String(v).trim() !== '');
    if (!staff) continue;

    const end = addMinutes(start, minutes);
    const label = `${fmt(start,'EEE dd MMM')} • ${fmt(start,'HH:mm')}–${fmt(end,'HH:mm')}`;
    const key = norm(String(staff));
    (tmp.get(key) ?? tmp.set(key, []).get(key)!).push(label);
  }

  const out = new Map<string,string>();
  for (const [k, arr] of tmp.entries()) out.set(k, arr.sort().join('; '));
  return out;
}

// Attach to your existing employee summary rows
export function attachCancelledVisits(
  summaryRows: Array<{ name: string; [k: string]: any }>,
  windowsByStaff: Map<string, string>,
  colKey = 'cancelledVisits'
) {
  for (const row of summaryRows) row[colKey] = windowsByStaff.get(norm(row.name)) ?? '—';
  return summaryRows;
}
