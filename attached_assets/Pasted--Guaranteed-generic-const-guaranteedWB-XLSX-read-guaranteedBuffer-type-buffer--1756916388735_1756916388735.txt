// ---- Guaranteed (generic) ----
const guaranteedWB = XLSX.read(guaranteedBuffer, { type: 'buffer' });
const guaranteedWS = pickBestSheet(guaranteedWB, [/data/i, /guaranteed/i]);
const gMatrix      = XLSX.utils.sheet_to_json<any[]>(guaranteedWS, { header: 1, raw: true }) as any[][];
const { headerRowIdx: gHdrRow, headers: gHeaders } = findHeaderRow(guaranteedWS);

const gSynonyms = {
  empName: ["Actual Employee Name", "Employee Name", "Carer Name", "Care Pro Name", "Name"],
  weekly:  ["Actual Employee Hours Per Week", "Weekly Hours", "Contract Hours", "Hours per Week"],
  pay:     ["Actual Pay Rate Hours", "Pay Rate Hours", "Pay Hours"],
  startDT: ["Service Requirement Start Date And Time", "Start Date Time", "Start Date", "Planned Start Date And Time"],
  endDT:   ["Service Requirement End Date And Time", "End Date Time", "End Date", "Planned End Date And Time"],
  stype:   ["Actual Service Type Description", "Service Type Description", "Service Type", "Planned Service Type Description"],
  cancel:  ["Cancellation Description", "Cancellation", "Cancelled Reason"]
};
const gMap = mapCols(gHeaders, gSynonyms);
const gIdx: Record<string, number> = {};
gHeaders.forEach((h, i) => (gIdx[h] = i));

const guaranteedData: GuaranteedHoursRow[] = [];
for (const row of gMatrix.slice(gHdrRow + 1)) {
  const get = (k: keyof typeof gMap) => (gMap[k] ? row[gIdx[gMap[k] as string]] : undefined);
  const empName = get("empName");
  const weekly  = Number(get("weekly"));
  const pay     = Number(get("pay"));
  const startDT = get("startDT");
  const endDT   = get("endDT");
  if (!empName || !weekly || !pay || !startDT || !endDT) continue;

  const serviceType = String(get("stype") ?? "");
  const cancellation = String(get("cancel") ?? "").trim();
  // Apply SAME rules here:
  const isSecondary = serviceType === "Multiple Care (Secondary)";
  const isCancelled = cancellation.length > 0;
  if (isSecondary || isCancelled) continue;

  guaranteedData.push({
    "Actual Employee Name": String(empName),
    "Actual Employee Hours Per Week": weekly,
    "Actual Pay Rate Hours": pay,
    "Service Requirement Start Date And Time": startDT,
    "Service Requirement End Date And Time": endDT,
    "Actual Service Type Description": serviceType,
    "Cancellation Description": cancellation || undefined
  } as any);
}
