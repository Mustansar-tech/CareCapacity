// ====================== HELPERS ======================

function parseHours(startHHMM, endHHMM) {
  if (!startHHMM || !endHHMM) return 0;
  const [sh, sm] = startHHMM.split(":").map(Number);
  const [eh, em] = endHHMM.split(":").map(Number);
  const start = sh + sm/60;
  const end   = eh + em/60;
  return Math.max(0, end - start);
}
function sumBy(arr, fn) { return arr.reduce((acc, r) => acc + (fn(r) || 0), 0); }
function classDelta(v){ return v < 0 ? styles.negative : v > 0 ? styles.positive : styles.neutral; }

function aggregateAvailabilityForWeek(availability, sickness, holidays, clientDemandDaily, weekDates) {
  const availByDate = new Map();  // date -> available hrs total
  const sickByDate  = new Map();
  const holByDate   = new Map();

  // Availability
  availability.forEach(a => {
    const hours = ("slotHours" in a && a.slotHours!=null) ? a.slotHours : parseHours(a.slotStart, a.slotEnd);
    availByDate.set(a.date, (availByDate.get(a.date)||0) + hours);
  });

  // Sickness/Holidays
  sickness.forEach(s => sickByDate.set(s.date, (sickByDate.get(s.date)||0) + (s.hours||0)));
  holidays.forEach(h => holByDate.set(h.date, (holByDate.get(h.date)||0) + (h.hours||0)));

  // Client demand map
  const cm = new Map(clientDemandDaily.map(d => [d.date, d.client_hours]));
  const allDates = weekDates && weekDates.length ? weekDates
                   : Array.from(new Set([
                       ...availability.map(a=>a.date),
                       ...sickness.map(s=>s.date),
                       ...holidays.map(h=>h.date),
                       ...clientDemandDaily.map(c=>c.date),
                     ])).sort();

  return allDates.map(date => {
    const avail = +(availByDate.get(date)||0).toFixed(2);
    const sick  = +(sickByDate.get(date)||0).toFixed(2);
    const hol   = +(holByDate.get(date)||0).toFixed(2);
    const net   = Math.max(0, +(avail - sick - hol).toFixed(2));
    const client = +(cm.get(date)||0);
    const delta = +(net - client).toFixed(2);
    return { date, avail, sickness: sick, holiday: hol, net, client, delta };
  });
}

function buildEmployeeDayRows(date, employees, availability, sickness, holidays, assigned) {
  const avForDay = availability.filter(a => a.date === date);
  const sickForDay = sickness.filter(s => s.date === date);
  const holForDay = holidays.filter(h => h.date === date);
  const assignedForDay = assigned.filter(a => a.date === date);

  const byEmpAvail = new Map();
  avForDay.forEach(a => {
    const hours = ("slotHours" in a && a.slotHours!=null) ? a.slotHours : parseHours(a.slotStart, a.slotEnd);
    const arr = byEmpAvail.get(a.employeeId) || [];
    arr.push({ start: a.slotStart, end: a.slotEnd, hours });
    byEmpAvail.set(a.employeeId, arr);
  });

  const sickMap = new Map(sickForDay.map(s => [s.employeeId, s.hours||0]));
  const holMap  = new Map(holForDay.map(h => [h.employeeId,  h.hours||0]));
  const asgMap  = new Map(assignedForDay.map(a => [a.employeeId, a.hours||0]));

  const rows = employees.map(e => {
    const slots = byEmpAvail.get(e.id) || [];
    const availableHours = +(sumBy(slots, s => s.hours).toFixed(2));
    const slotWindows = slots.map(s => `${s.start}-${s.end}`).join("; ");
    const sickHrs = +(sickMap.get(e.id)||0);
    const holHrs = +(holMap.get(e.id)||0);
    const net = Math.max(0, +(availableHours - sickHrs - holHrs).toFixed(2));
    const assignedHrs = +(asgMap.get(e.id)||0);

    const wdCount = e.workingDays?.length || 5;
    const contractedDaily = +( (e.contractedWeeklyHours / wdCount) || 0 ).toFixed(2);
    const remaining = Math.max(0, +(Math.min(net, Math.max(0, contractedDaily - assignedHrs))).toFixed(2));

    return {
      employeeId: e.id,
      name: e.name,
      contractedWeeklyHours: e.contractedWeeklyHours || 0,
      workingDays: e.workingDays || [],
      slotWindows,
      availableHours,
      sickHrs,
      holHrs,
      net,
      assignedHrs,
      contractedDaily,
      remaining,
      suggested: 0,
      notes: ""
    };
  });

  return rows.filter(r => r.availableHours > 0 || r.remaining > 0);
}

// ====================== EXCEL UPLOAD & TRANSFORM ======================

/**
 * Expected sheets:
 *  - "CAREGiver Availability": CAREGiver Name, Start Date, Start Time, End Date, End Time, Hours, Type
 *  - "Client Demand": Date, Client Hours
 * This function parses the workbook and returns:
 *  { employees, availability, sickness, holidays, assigned, clientDemandDaily, weekDates }
 */
function parseWorkbook(file) {
  return new Promise((resolve, reject) => {
    const reader = new FileReader();
    reader.onerror = () => reject(new Error("Failed to read file"));
    reader.onload = (e) => {
      try {
        const data = new Uint8Array(e.target.result);
        const wb = XLSX.read(data, { type: "array" });

        const availSheet = wb.Sheets["CAREGiver Availability"];
        if (!availSheet) throw new Error("Missing sheet: 'CAREGiver Availability'");
        const availJson = XLSX.utils.sheet_to_json(availSheet, { defval: "" });

        const demandSheet = wb.Sheets["Client Demand"];
        if (!demandSheet) throw new Error("Missing sheet: 'Client Demand'");
        const demandJson = XLSX.utils.sheet_to_json(demandSheet, { defval: "" });

        // Normalize headers (lowercase, strip spaces)
        const norm = (s) => String(s||"").toLowerCase().replace(/\s+/g,"").trim();

        // Validate availability columns
        const aCols = Object.keys(availJson[0] || {}).map(norm);
        const reqA = ["caregivername","startdate","starttime","enddate","endtime","hours","type"];
        const missingA = reqA.filter(c => !aCols.includes(c));
        if (missingA.length) throw new Error(`'CAREGiver Availability' missing columns: ${missingA.join(", ")}`);

        // Validate demand columns
        const dCols = Object.keys(demandJson[0] || {}).map(norm);
        const reqD = ["date","clienthours"];
        const missingD = reqD.filter(c => !dCols.includes(c));
        if (missingD.length) throw new Error(`'Client Demand' missing columns: ${missingD.join(", ")}`);

        // Build employees map (id by name for now; later replace with Employee ID)
        const empIdByName = new Map();
        const employees = [];
        function ensureEmp(name) {
          const n = String(name||"").trim();
          if (!n) return null;
          if (!empIdByName.has(n)) {
            const id = "EMP_" + (employees.length+1);
            empIdByName.set(n, id);
            // ContractedWeeklyHours & WorkingDays not in PP export — default placeholders
            employees.push({ id, name: n, contractedWeeklyHours: 0, workingDays: ["Mon","Tue","Wed","Thu","Fri"] });
          }
          return empIdByName.get(n);
        }

        // Availability, Sickness, Holidays
        const availability = [];
        const sickness = [];
        const holidays = [];
        const assigned = []; // not from PP; keep empty for now

        const toISO = (v) => {
          // Accept dd/mm/yyyy or excel date serial or yyyy-mm-dd
          if (typeof v === "number") {
            // Excel serial
            const d = XLSX.SSF.parse_date_code(v);
            const dt = dayjs(new Date(Date.UTC(d.y, d.m-1, d.d)));
            return dt.format("YYYY-MM-DD");
          }
          const s = String(v||"").trim();
          if (!s) return "";
          const tryDMY = s.match(/^(\d{1,2})[\/\-](\d{1,2})[\/\-](\d{2,4})$/);
          if (tryDMY) {
            const [_, dd, mm, yyyy] = tryDMY;
            const yr = (yyyy.length===2 ? ("20"+yyyy) : yyyy);
            return dayjs(`${yr}-${mm.padStart(2,"0")}-${dd.padStart(2,"0")}`).format("YYYY-MM-DD");
          }
          // else assume ISO already
          return dayjs(s).format("YYYY-MM-DD");
        };

        const toHHMM = (v) => {
          if (typeof v === "number") {
            // Excel time as fraction of a day
            const totalMins = Math.round(v * 24 * 60);
            const hh = String(Math.floor(totalMins/60)).padStart(2,"0");
            const mm = String(totalMins%60).padStart(2,"0");
            return `${hh}:${mm}`;
          }
          const s = String(v||"").trim();
          if (!s) return "";
          const m = s.match(/^(\d{1,2}):(\d{2})/);
          if (m) return `${m[1].padStart(2,"0")}:${m[2]}`;
          return s;
        };

        availJson.forEach(row => {
          // Normalize
          // Original keys might be: "CAREGiver Name", "Start Date", "Start Time", etc.
          const name = row["CAREGiver Name"] ?? row["Caregiver Name"] ?? row["CAREGiver name"] ?? row["Name"];
          const type = String(row["Type"] ?? "").trim();
          const startDate = toISO(row["Start Date"]);
          const endDate   = toISO(row["End Date"]);
          const startTime = toHHMM(row["Start Time"]);
          const endTime   = toHHMM(row["End Time"]);
          const hours     = Number(row["Hours"] ?? 0);

          const empId = ensureEmp(name);
          if (!empId || !startDate) return;

          const date = startDate; // PP export uses separate end date; assume same-day ranges (common case)
          const slot = { employeeId: empId, date, slotStart: startTime, slotEnd: endTime };
          if (type.toLowerCase() === "available") {
            availability.push(slot);
          } else if (type.toLowerCase() === "sickness" || type.toLowerCase() === "sick") {
            sickness.push({ employeeId: empId, date, hours: hours || parseHours(startTime, endTime) });
          } else if (type.toLowerCase() === "holiday" || type.toLowerCase() === "holidays") {
            holidays.push({ employeeId: empId, date, hours: hours || parseHours(startTime, endTime) });
          } else {
            // ignore other types for now
          }
        });

        const clientDemandDaily = demandJson.map(r => ({
          date: toISO(r["Date"]),
          client_hours: Number(r["Client Hours"] || 0)
        })).filter(r => r.date);

        const weekDates = Array.from(new Set([
          ...availability.map(a=>a.date),
          ...sickness.map(s=>s.date),
          ...holidays.map(h=>h.date),
          ...clientDemandDaily.map(c=>c.date),
        ])).sort();

        resolve({ employees, availability, sickness, holidays, assigned, clientDemandDaily, weekDates });
      } catch (err) {
        reject(err);
      }
    };
    reader.readAsArrayBuffer(file);
  });
}

function DailyDetail({ date, setDate, joined, employees, availability, sickness, holidays, assigned }) {
  const day = joined.find(d => d.date === date) || joined[0];
  const dates = joined.map(d => d.date);
  const [rows, setRows] = useState(() => buildEmployeeDayRows(day.date, employees, availability, sickness, holidays, assigned));

  useEffect(() => {
    setRows(buildEmployeeDayRows(date, employees, availability, sickness, holidays, assigned));
  }, [date, employees, availability, sickness, holidays, assigned]);

  const [sortKey, setSortKey] = useState("remaining");
  const [q, setQ] = useState("");

  const sortedFiltered = useMemo(() => {
    const f = rows.filter(r => r.name.toLowerCase().includes(q.toLowerCase()));
    const s = [...f].sort((a,b) => sortKey==="name" ? a.name.localeCompare(b.name) : (b[sortKey]||0)-(a[sortKey]||0));
    return s;
  }, [rows, sortKey, q]);

  const totals = useMemo(() => ({
    net: sumBy(sortedFiltered, r=>r.net),
    assigned: sumBy(sortedFiltered, r=>r.assignedHrs),
    remaining: sumBy(sortedFiltered, r=>r.remaining),
    suggested: sumBy(sortedFiltered, r=>r.suggested),
  }), [sortedFiltered]);

  function setSuggested(idx, val) {
    setRows(prev => {
      const next = [...prev];
      const v = Math.max(0, Math.min(Number(val||0), next[idx].remaining));
      next[idx] = { ...next[idx], suggested: v };
      return next;
    });
  }

  function exportVisibleCSV() {
    const header = [
      "Date","Employee","Contracted Weekly Hrs","Working Days","Contracted Daily Hrs",
      "Availability Windows","Available Hrs","Sick Hrs","Holiday Hrs","Net Available Hrs",
      "Assigned Hrs","Remaining Assignable Hrs","Suggested Allocation","Notes"
    ];
    const body = sortedFiltered.map(r => [
      date, r.name, r.contractedWeeklyHours, r.workingDays.join(" "), r.contractedDaily,
      r.slotWindows || "-", r.availableHours, r.sickHrs, r.holHrs, r.net,
      r.assignedHrs, r.remaining, r.suggested || 0, r.notes || ""
    ]);
    const csv = Papa.unparse([header, ...body]);
    const blob = new Blob([csv], { type: "text/csv;charset=utf-8;" });
    const url = URL.createObjectURL(blob);
    const a = document.createElement("a");
    a.href = url; a.download = `day_sheet_${date}.csv`; a.click();
    URL.revokeObjectURL(url);
  }

  return (
    <div style={styles.stack}>
      <div style={{ display: "flex", gap: 12, alignItems: "center", flexWrap: "wrap" }}>
        <label style={styles.muted}>Select day</label>

              ))}

            <tfoot>
              <tr>
                <td colSpan={8} style={{ textAlign: "right", fontWeight: 600 }}>Totals:</td>
                <td><b>{totals.net.toFixed(1)}</b></td>
                <td>{totals.assigned.toFixed(1)}</td>
                <td className={classDelta(totals.remaining)}><b>{totals.remaining.toFixed(1)}</b></td>
                <td><b>{totals.suggested.toFixed(1)}</b></td>
                <td></td>
              </tr>
            </tfoot>
          </table>
        </div>
      </div>
    </div>
  );
}

function ExportPreview({ joined }) {
  const header = ["date","avail","sickness","holiday","net","client","delta"];
  const rows = joined.map(r => [r.date, r.avail, r.sickness, r.holiday, r.net, r.client, r.delta]);
  const csv = [header.join(","), ...rows.map(r=>r.join(","))].join("\n");

  function downloadCSV(){
    const blob = new Blob([csv], { type: "text/csv;charset=utf-8;" });
    const url = URL.createObjectURL(blob);
    const a = document.createElement("a");
    a.href = url; a.download = `capacity_export_${new Date().toISOString().slice(0,10)}.csv`; a.click();
    URL.revokeObjectURL(url);
  }

  return (
    <div style={styles.stack}>
      <div style={styles.panel}>
        <div style={styles.panelTitle}>Export columns (Scheduling)</div>
        <pre style={styles.pre}>{csv}</pre>
        <button style={styles.btn} onClick={downloadCSV}>Download CSV</button>
      </div>
      <div style={styles.panel}>
        <p style={styles.muted}>When client time bands are available, extend export to include band columns and split Net by overlap.</p>
      </div>
    </div>
  );
}

function UploadPanel({ onLoadData, onUseMocks, usingMocks }) {
  const [report, setReport] = useState("");

  async function onFile(e){
    const file = e.target.files?.[0];
    if (!file) return;
    setReport("Parsing workbook…");
    try {
      const parsed = await parseWorkbook(file);
      onLoadData(parsed);
      setReport(`Loaded ${parsed.weekDates.length} day(s), ${parsed.employees.length} employees, ${parsed.availability.length} availability slots.`);
    } catch (err) {
      console.error(err);
      setReport("Error: " + err.message);
    }
  }

  return (
    <div style={styles.stack}>
      <div style={styles.panel}>
        <div style={styles.panelTitle}>Upload Excel (People Planner + Client Demand)</div>
        <ol style={{ paddingLeft: 18, lineHeight: 1.6 }}>
          <li>Sheet <b>“CAREGiver Availability”</b> with columns: CAREGiver Name, Start Date, Start Time, End Date, End Time, Hours, Type</li>
          <li>Sheet <b>“Client Demand”</b> with columns: Date, Client Hours</li>
        </ol>
        <input type="file" accept=".xlsx,.xls" onChange={onFile} style={{ marginTop: 8 }} />
        <div style={{ marginTop: 8 }}>
          <button style={styles.btnGhost} onClick={onUseMocks}>
            {usingMocks ? "Mocks are ON" : "Switch to Mock Data"}
          </button>
        </div>
        <div style={{ marginTop: 10, fontSize: 13, color: "#475569" }}>{report}</div>
      </div>

      <div style={styles.panel}>
        <div style={styles.panelTitle}>Notes</div>
        <ul style={{ paddingLeft: 18, lineHeight: 1.6 }}>
          <li>“Available” rows create availability slots. “Sickness” / “Holiday” rows deduct hours from that day.</li>
          <li>Employees are identified by name in this version (Employee ID can be added later).</li>
          <li>Contracted weekly hours aren’t in the People Planner export; placeholders are set to 0 for uploaded data (BD view still works; Scheduling table shows 0 contracted daily hours unless you enrich later).</li>
        </ul>
      </div>
    </div>
  );
}

// ====================== APP ROOT ======================

export default function App() {
  // Data state (starts with mocks)
  const [usingMocks, setUsingMocks] = useState(true);
  const [employees, setEmployees] = useState(MOCK_employees);
  const [availability, setAvailability] = useState(MOCK_availability);
  const [sickness, setSickness] = useState(MOCK_sickness);
  const [holidays, setHolidays] = useState(MOCK_holidays);
  const [assigned, setAssigned] = useState(MOCK_assigned);
  const [clientDemandDaily, setClientDemandDaily] = useState(MOCK_clientDemandDaily);
  const [weekDates, setWeekDates] = useState(DEFAULT_WEEK);

  function loadFromParsed({ employees, availability, sickness, holidays, assigned, clientDemandDaily, weekDates }) {
    setUsingMocks(false);
    setEmployees(employees);
    setAvailability(availability);
    setSickness(sickness);
    setHolidays(holidays);
    setAssigned(assigned || []);
    setClientDemandDaily(clientDemandDaily);
    setWeekDates(weekDates.length ? weekDates : DEFAULT_WEEK);
    setActiveTab("overview");
  }

  function switchToMocks() {
    setUsingMocks(true);
    setEmployees(MOCK_employees);
    setAvailability(MOCK_availability);
    setSickness(MOCK_sickness);
    setHolidays(MOCK_holidays);
    setAssigned(MOCK_assigned);
    setClientDemandDaily(MOCK_clientDemandDaily);
    setWeekDates(DEFAULT_WEEK);
  }

  const [activeTab, setActiveTab] = useState("overview");

  const joined = useMemo(
    () => aggregateAvailabilityForWeek(availability, sickness, holidays, clientDemandDaily, weekDates),
    [availability, sickness, holidays, clientDemandDaily, weekDates]
  );

  const totals = useMemo(() => ({
    net: sumBy(joined, r=>r.net),
    client: sumBy(joined, r=>r.client),
    delta: sumBy(joined, r=>r.net) - sumBy(joined, r=>r.client),
    sickness: sumBy(joined, r=>r.sickness),
    holiday: sumBy(joined, r=>r.holiday),
  }), [joined]);

  const [selectedDate, setSelectedDate] = useState(joined[0]?.date || weekDates[0]);
  useEffect(() => {
    if (joined.length) setSelectedDate(joined[0].date);
  }, [joined]);

  return (
    <div style={styles.app}>
      <div style={styles.topbar}>
        <div style={styles.brand}>Care Hours vs Client Demand</div>
        <div style={{ marginLeft: "auto", display: "flex", gap: 8, flexWrap: "wrap" }}>
          <TabButton label="Overview" active={activeTab==="overview"} onClick={()=>setActiveTab("overview")} />
          <TabButton label="Daily Detail" active={activeTab==="daily"} onClick={()=>setActiveTab("daily")} />
          <TabButton label="Sickness & Holidays" active={activeTab==="impact"} onClick={()=>setActiveTab("impact")} />
          <TabButton label="Export" active={activeTab==="export"} onClick={()=>setActiveTab("export")} />
          <TabButton label="Upload & Data" active={activeTab==="upload"} onClick={()=>setActiveTab("upload")} />
        </div>
      </div>

      <main style={styles.main}>
        {activeTab === "overview" && <Overview joined={joined} totals={totals} onPickDate={(d)=>{ setSelectedDate(d); setActiveTab("daily"); }} />}
        {activeTab === "daily"    && <DailyDetail date={selectedDate} setDate={setSelectedDate} joined={joined}
                                                   employees={employees} availability={availability}
                                                   sickness={sickness} holidays={holidays} assigned={assigned} />}
        {activeTab === "impact"   && <Impact joined={joined} />}
        {activeTab === "export"   && <ExportPreview joined={joined} />}
        {activeTab === "upload"   && <UploadPanel onLoadData={loadFromParsed} onUseMocks={switchToMocks} usingMocks={usingMocks} />}
      </main>
    </div>
  );
}

      {label}
    </button>
  );
}
