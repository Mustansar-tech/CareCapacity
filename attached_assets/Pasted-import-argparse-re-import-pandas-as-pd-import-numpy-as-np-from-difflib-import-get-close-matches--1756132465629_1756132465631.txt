import argparse, re
import pandas as pd
import numpy as np
from difflib import get_close_matches

LEAVE = ["Maternity/Paternity","Sick","Holiday","Compassionate Leave","Other Unavailable","Pre-Agreed Appointment"]
PRIORITY = {"Maternity/Paternity":1,"Sick":2,"Holiday":3,"Compassionate Leave":4,"Other Unavailable":5,"Pre-Agreed Appointment":6,"Available":7}

def normalize_name(s):
    if pd.isna(s): return ""
    s = str(s).lower()
    s = re.sub(r"\(.*?\)", "", s)
    s = re.sub(r"[^a-z\s]", " ", s)
    s = re.sub(r"\b(mr|mrs|miss|ms|dr)\b", " ", s)
    s = re.sub(r"\s+", " ", s).strip()
    return " ".join(sorted(s.split()))

def tstr(x):
    try: return pd.to_datetime(x).strftime("%H:%M")
    except: return ""

def hours_between(a,b):
    try:
        st, en = pd.to_datetime(a), pd.to_datetime(b)
        d = (en-st).total_seconds()/3600.0
        return round(d if d>=0 else d+24, 2)
    except: return np.nan

def run(av_path, gh_path, out_path):
    avail = pd.read_excel(av_path, sheet_name="CAREGiver Availability")
    gh    = pd.read_excel(gh_path, sheet_name="Data")

    gh = gh[['Actual Employee Name','Actual Employee Hours Per Week']].dropna(how='all').rename(
        columns={'Actual Employee Name':'Employee Name','Actual Employee Hours Per Week':'Contracted Weekly Hours'})
    gh['gh_key'] = gh['Employee Name'].map(normalize_name)

    avail = avail.rename(columns={'CAREGiver Name':'Employee Name','Start Date':'Date','Type':'Status','Notes':'Notes'})
    avail['Date'] = pd.to_datetime(avail['Date'], dayfirst=True, errors="coerce").dt.date
    avail['Time Windows'] = avail['Start Time'].map(tstr)+"-"+avail['End Time'].map(tstr)
    avail['name_key'] = avail['Employee Name'].map(normalize_name)

    keys = gh['gh_key'].unique().tolist()
    avail['gh_key'] = avail['name_key'].apply(lambda k: get_close_matches(k, keys, n=1, cutoff=0.7)[0] if k else None)
    avail = avail[avail['gh_key'].notna()]

    m = avail.merge(gh[['gh_key','Employee Name','Contracted Weekly Hours']], on='gh_key', how='inner', suffixes=('','_gh'))
    m['Employee Name'] = m['Employee Name_gh'].fillna(m['Employee Name']); m = m.drop(columns=['Employee Name_gh'])

    days = m.groupby('gh_key')['Date'].nunique().rename('Days Available')
    m = m.merge(days, on='gh_key', how='left')
    m['Contracted Daily Hours'] = (m['Contracted Weekly Hours']/m['Days Available']).round(2)

    m['Hours_calc'] = m.apply(lambda r: hours_between(r['Start Time'], r['End Time']), axis=1)
    m['Hours_effective'] = np.where(m['Hours'].notna(), m['Hours'], m['Hours_calc'])

    def collapse(g):
        emp = g['Employee Name'].iloc[0]
        wk  = float(g['Contracted Weekly Hours'].iloc[0])
        day = float(g['Contracted Daily Hours'].iloc[0]) if pd.notna(g['Contracted Daily Hours'].iloc[0]) else 0.0
        dt  = g['Date'].iloc[0]
        d = g.copy()
        d['Start_str']=d['Start Time'].map(tstr); d['End_str']=d['End Time'].map(tstr)
        d = d.drop_duplicates(subset=['Status','Start_str','End_str'])
        agg = d.groupby('Status', dropna=False).agg(
            Hours_raw=('Hours_effective','sum'),
            Windows=('Time Windows', lambda s: "; ".join(sorted(set([x for x in s if isinstance(x,str) and x not in ("","-")])))),
            Notes=('Notes', lambda s: "; ".join(sorted(set([x for x in s.dropna().astype(str) if x!=""])))),
        ).reset_index()
        leave_raw = agg[agg['Status'].isin(LEAVE)]['Hours_raw'].sum()
        leave_cap = min(leave_raw, day)
        rows=[]
        for _,r in agg.iterrows():
            st, tw, nts = r['Status'], r['Windows'], r['Notes']
            if st=="Available":
                hours = max(day-leave_cap, 0.0); net = hours
            elif st in LEAVE:
                hours = min(float(r['Hours_raw'] or 0.0), day); net = 0.0
            else:
                hours = float(r['Hours_raw'] or 0.0); net = 0.0
            rows.append({"Employee Name":emp,"Contracted Weekly Hours":round(wk,2),
                         "Contracted Daily Hours":round(day,2),"Date":dt,"Status":st,
                         "Time Windows":tw,"Hours":round(hours,2),"Net Capacity":round(net,2),"Notes":nts})
        out = pd.DataFrame(rows); out["Priority"]=out["Status"].map(PRIORITY).fillna(999)
        return out.sort_values("Priority")

    cleaned = m.groupby(['gh_key','Date'], group_keys=False).apply(collapse).reset_index(drop=True)
    cleaned.to_excel(out_path, index=False)
    print(f"Saved {out_path} ({len(cleaned)} rows)")

if __name__=="__main__":
    p=argparse.ArgumentParser()
    p.add_argument("--availability","-a",required=True)
    p.add_argument("--guaranteed","-g",required=True)
    p.add_argument("--out","-o",default="cleaned_capacity.xlsx")
    args=p.parse_args()
    run(args.availability,args.guaranteed,args.out)
