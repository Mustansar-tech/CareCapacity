from playwright.sync_api import sync_playwright

PP_URL = "https://www.peopleplanner.biz/Security/Login.aspx"

print("Starting People Planner – stable single-page mode")

with sync_playwright() as p:
    browser = p.chromium.launch(
        channel="msedge",
        headless=False
    )

    context = browser.new_context()

    # 🔒 HARD BLOCK: close any extra pages immediately
    context.on("page", lambda new_page: new_page.close())

    page = context.new_page()

    page.goto(PP_URL, wait_until="domcontentloaded")

    print("➡ Log in manually")
    print("➡ Wait until dashboard is fully clickable")
    input("Press ENTER here when ready to close...")

    browser.close()
